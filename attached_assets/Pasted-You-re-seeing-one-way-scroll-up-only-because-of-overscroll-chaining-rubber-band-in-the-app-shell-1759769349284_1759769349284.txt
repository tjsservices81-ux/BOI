Apply the fix: bidirectional scroll lock on Processing/Success/Back

Goal: When on TransferProcessing, TransferSuccess, and during the Dashboard return transition, the page must not move up or down. Only designated inner panes (none on these screens) may scroll. Works in PWA + browser.

What to change
	1.	CSS (global) — add a lock class for the app shell (not body)

/* app-shell scroll lock */
.app-shell.locked,
#app.locked, #root.locked {
  height: 100dvh;           /* stable viewport in PWA */
  overflow: hidden;         /* blocks up & down */
  overscroll-behavior: none;/* stops rubber-band/pull-to-refresh */
  touch-action: none;       /* blocks touch-scroll + inertial */
}

/* Safety: ensure inner panels don’t re-enable scroll here */
.app-shell.locked * {
  overscroll-behavior: contain;
}

	2.	App shell markup — ensure your top wrapper has .app-shell (or id="app"/root"). Example:

<div id="app" class="app-shell"> ... </div>

	3.	Route-scoped lock helper (React) — add a small utility and use it on the three screens.

Create src/utils/useScrollLock.ts (or .js):

import { useEffect } from 'react';

export function useAppScrollLock(locked: boolean) {
  useEffect(() => {
    const el = document.querySelector('.app-shell') || 
               document.getElementById('app') || 
               document.getElementById('root');
    if (!el) return;

    if (locked) {
      el.classList.add('locked');
      // prevent accidental document scroll position jumps
      document.documentElement.style.scrollBehavior = 'auto';
      window.scrollTo(0, 0);
    } else {
      el.classList.remove('locked');
      document.documentElement.style.scrollBehavior = '';
    }

    return () => {
      el?.classList.remove('locked');
      document.documentElement.style.scrollBehavior = '';
    };
  }, [locked]);
}

Use it in the screens:

// TransferProcessing.tsx
import { useAppScrollLock } from '@/utils/useScrollLock';
export default function TransferProcessing() {
  useAppScrollLock(true);
  // ...rest
}

// TransferSuccess.tsx
import { useAppScrollLock } from '@/utils/useScrollLock';
export default function TransferSuccess() {
  useAppScrollLock(true);
  // ...rest
}

	4.	Back to Dashboard transition — lock during the transition, then release once the dashboard is mounted.

// In the component handling navigation after success
import { useAppScrollLock } from '@/utils/useScrollLock';
export default function ReturnToDashboard() {
  useAppScrollLock(true); // lock while transitioning
  // when navigation completes, Dashboard will mount and unlock by default
  return null;
}

// In Dashboard.tsx ensure it DOES NOT lock:
import { useAppScrollLock } from '@/utils/useScrollLock';
export default function Dashboard() {
  useAppScrollLock(false);
  // ...dashboard content
}
