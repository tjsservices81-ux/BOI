(function () {
  const mode = window.matchMedia('(display-mode: standalone)').matches ? 'standalone' : 'browser';
  console.log('Display mode:', mode);
  const isScrollable = (el) => {
    const cs = getComputedStyle(el);
    const canScrollY = /(auto|scroll|overlay|clip)/.test(cs.overflowY);
    return canScrollY && el.scrollHeight > el.clientHeight;
  };
  const rows = [];
  document.querySelectorAll('*').forEach(el => {
    if (isScrollable(el)) {
      rows.push({
        node: el.tagName.toLowerCase() + (el.id ? `#${el.id}` : ''),
        classes: (el.className || '').toString().slice(0,120),
        overflowY: getComputedStyle(el).overflowY,
        scrollHeight: el.scrollHeight,
        clientHeight: el.clientHeight
      });
      el.style.outline = '2px dashed red';
    }
  });
  console.table(rows);
  return { mode, rows };
})();
